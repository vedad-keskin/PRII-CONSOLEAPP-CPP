#include <iostream>
using namespace std;

char* GetNizKaraktera(const char* tekst) {
	if (tekst == nullptr)
		return nullptr;
	int vel = strlen(tekst) + 1;
	char* novi = new char[vel];
	strcpy_s(novi, vel, tekst);
	return novi;
}

//Kolekcija<int, 10> kolekcijaBrojeva;
template<class T, int max>
class Kolekcija {
public:
	T* _niz;
	int _trenutno;


	Kolekcija() {

		_trenutno = 0;

		_niz = new T[max];

	}

	Kolekcija(const Kolekcija& obj) {

		_trenutno = obj._trenutno;

		_niz = new T[max];

		for (int i = 0; i < _trenutno; i++)
		{
			_niz[i] = obj._niz[i];
		}

	}

	Kolekcija& operator = (const Kolekcija& obj) {
		if (this != &obj) {
			delete[] _niz;

			_trenutno = obj._trenutno;

			_niz = new T[max];

			for (int i = 0; i < _trenutno; i++)
			{
				_niz[i] = obj._niz[i];
			}
		}
		return *this;
	}

	~Kolekcija() {

		delete[] _niz;
		_niz = nullptr;

	}

	friend ostream& operator << (ostream& COUT, const Kolekcija& obj) {

		
		for (int i = 0; i < obj._trenutno; i++)
		{
			COUT << obj._niz[i] << endl;
		}


		return COUT;
	}








	void AddElement(T el) { /* alt. Dodaj */
		/*
		* Dodaje element tipa T u niz
		* Onemoguciti izlazak niza izvan opsega bacajem izuzetka tipa exception
		* Onemoguciti dupliranje elemenata bacajem izuzetka tipa exception
		*/

		if (max == _trenutno) {

			throw exception("Niz je popunjen");

		}


		for (int i = 0; i < _trenutno; i++)
		{
			// int == int , 6 == 5 , char == char , A == B , Zivotinja == Zivotinja
			if (el == _niz[i]) {

				throw exception("Dupliranje nije dozvoljeno");

			}
		}



		_niz[_trenutno] = el;

		_trenutno++;


	}


	/*
	* metoda prihvata indekse lokacije, te vraca elemente koji se u kolekciji
	* nalaze na proslijedjenim lokacijama (ukljucujuci i te lokacije)
	*
	* Nakon izvrsenja, objekat kolekcijaOperator bi trebao posjedovati sljedeci sadrzaj:
	* C
	* D
	* E
	* F
	*
	* objekat kolekcijaKopija ostaje nepromijenjen
	*/
	//Kolekcija<char, 10> kolekcijaOperator = kolekcijaKopija(2, 5);

	Kolekcija operator() (int lokacijaOd, int lokacijaDo) {


		Kolekcija kolekcijaOperator;


		for (int i = lokacijaOd; i <= lokacijaDo; i++) // 2 3 4 5
		{

			kolekcijaOperator.AddElement(_niz[i]);

		}

		return kolekcijaOperator;

	}


	/*
	* metoda RemoveAt treba da ukloni i vrati (kao povratnu vrijednost)
	* clanove kolekcije koji se nalaze na lokaciji definisanoj prvim parametrom.
	*
	* Nakon izvrsenja metode RemoveAt, objekat kolekcijaKopija bi trebao
	* posjedovati sljedeci sadrzaj:
	* A
	* B
	* D
	* E
	* ....
	*
	* a objekat kolekcijaRemoveAt samo element:
	* C
	*/
	//Kolekcija<char, 10> kolekcijaRemoveAt = kolekcijaKopija.RemoveAt(2);
	Kolekcija RemoveAt(int lokacija) { /* alt. UkloniSaPozicije */

		Kolekcija kolekcijaRemoveAt;

		Kolekcija kolekcijaPreostali;

		for (int i = 0; i < _trenutno; i++) // 0 1 2 3 4 .. 
		{

			if (lokacija == i) { // 2 == 2

				kolekcijaRemoveAt.AddElement(_niz[i]);

			}
			else {

				kolekcijaPreostali.AddElement(_niz[i]);

			}

		}

		*this = kolekcijaPreostali;

		return kolekcijaRemoveAt;

	}


	/*
	* InsertAt dodaje novi element na lokaciju definisanu prvim parametrom.
	* Metoda vraca novu kolekciju sa umetnutim elementom.
	* U slucaju popunjene kolekcije ili neispravne lokacije potrebno je baciti
	* izuzetak (za potrebe testiranja mozete dodati try catch blok).
	*
	* Nakon izvrsenja, objekat kolekcijaInsertAt bi trebao posjedovati sljedeci sadrzaj:
	* A
	* B
	* X
	* C
	* D
	* E
	* F
	* G
	* H
	* I
	*
	* objekat kolekcijaOriginal ostaje nepromijenjen
	*/
	//Kolekcija<char, 10> kolekcijaInsertAt = kolekcijaOriginal.InsertAt(2, 'X');
	Kolekcija InsertAt(int lokacija, T el) { /* alt. DodajNaPoziciju */

		//if (_trenutno == max) {

		//	throw exception("Niz je popunjen");

		//}

		if (lokacija < 0 || lokacija >= _trenutno) {

			throw exception("Lokacija nije ispravna");

		}


		Kolekcija kolekcijaInsertAt;

		for (int i = 0; i < _trenutno; i++) // 0 1 2 3 .. 
		{

			if (i == lokacija) { // 2 == 2

				kolekcijaInsertAt.AddElement(el);

			}

			kolekcijaInsertAt.AddElement(_niz[i]);

		}

		return kolekcijaInsertAt;

	}


	/*
	* RemoveRange uklanja broj elemenata definisan drugim parametrom, pocevsi od
	* lokacije definisane prvim parametrom (ukljucujuci tu lokaciju).
	* Metoda vraca pokazivac na novu kolekciju sa uklonjenim elementima,
	* a pozivalac je odgovoran za njenu dealokaciju.
	* U slucaju neispravnog opsega potrebno je baciti izuzetak.
	*
	* Nakon izvrsenja, objekat kolekcijaRemoveRange bi trebao posjedovati:
	* D
	* E
	*
	* a objekat kolekcijaOriginal:
	* A
	* B
	* C
	* F
	* G
	* H
	* I
	*/
	//Kolekcija<char, 10>* kolekcijaRemoveRange = kolekcijaOriginal.RemoveRange(3, 2);
	Kolekcija* RemoveRange(int lokacijaOd, int brojElementa) { /* alt. UkloniRaspon */

		if (lokacijaOd < 0 || lokacijaOd + brojElementa > _trenutno) {

			throw exception("Opseg nije validan");

		}


		Kolekcija* kolekcijaRemoveRange = new Kolekcija();

		Kolekcija kolekcijaPreostali;

		for (int i = 0; i < _trenutno; i++) // 0 1 2 3 4 .. 
		{

			//  i >= 3 ...      3 ... 4
			if (i >= lokacijaOd && i < lokacijaOd + brojElementa) { //

				kolekcijaRemoveRange->AddElement(_niz[i]);

			}
			else {

				kolekcijaPreostali.AddElement(_niz[i]);

			}

		}

		*this = kolekcijaPreostali;

		return kolekcijaRemoveRange;


	}

};


class Zivotinja {
	char* _naziv;
	int _tezina;
public:

	Zivotinja(const char* naziv = "---", int tezina = 0) {

		_naziv = GetNizKaraktera(naziv);
		_tezina = tezina;

	}

	Zivotinja(const Zivotinja& obj) {


		_naziv = GetNizKaraktera(obj._naziv);
		_tezina = obj._tezina;

	}

	Zivotinja& operator = (const Zivotinja& obj) {
		if (this != &obj) {

			delete[] _naziv;

			_naziv = GetNizKaraktera(obj._naziv);
			_tezina = obj._tezina;
		}
		return *this;
	}

	bool operator == (const Zivotinja& obj) {

		return strcmp(_naziv, obj._naziv) == 0;

	}

	~Zivotinja() {

		delete[] _naziv;
		_naziv = nullptr;

	}

	friend ostream& operator << (ostream& COUT, const Zivotinja& obj) {
		COUT << obj._naziv << " [" << obj._tezina << "kg]";
		return COUT;
	}



};



void main() {

	/* ========== KOLEKCIJA INT ========== */

	Kolekcija<int, 10> kolekcijaBrojeva;

	for (int i = 0; i < 10; i++)
	{
		kolekcijaBrojeva.AddElement(i); // Dodaj
	}

	cout << kolekcijaBrojeva << endl;

	try {
		/* AddElement - kapacitet popunjen (max == 10) */
		kolekcijaBrojeva.AddElement(99);
	}
	catch (exception& err) {
		cout << "Exception AddElement (kapacitet): " << err.what() << endl;
	}


	/* ========== KOLEKCIJA CHAR + AddElement / duplikat ========== */

	Kolekcija<char, 10> kolekcijaOriginal;

	/* 65 -> A , 66 -> B */
	int ascii = 65;

	for (int i = 0; i < 9; i++)
	{
		/* A B C D E F G H I */
		kolekcijaOriginal.AddElement(ascii + i);
	}

	try {
		/* AddElement - duplikat */
		kolekcijaOriginal.AddElement('C');
	}
	catch (exception& err) {
		cout << "Exception AddElement (duplikat): " << err.what() << endl;
	}

	cout << "ORIGINALNA KOLEKCIJA: " << endl;
	cout << kolekcijaOriginal << endl;

	/* konstruktor kopije */
	Kolekcija<char, 10> kolekcijaKopija = kolekcijaOriginal;

	cout << "KOLEKCIJA KOPIJA (ctor): " << endl;
	cout << kolekcijaKopija << endl;

	/* operator = */
	Kolekcija<char, 10> kolekcijaDodjela;
	kolekcijaDodjela = kolekcijaOriginal;

	cout << "KOLEKCIJA DODJELA (op=): " << endl;
	cout << kolekcijaDodjela << endl;

	/*
	* metoda prihvata indekse lokacije, te vraca elemente koji se u kolekciji
	* nalaze na proslijedjenim lokacijama (ukljucujuci i te lokacije)
	*
	* Nakon izvrsenja, objekat kolekcijaOperator bi trebao posjedovati sljedeci sadrzaj:
	* C
	* D
	* E
	* F
	*
	* objekat kolekcijaKopija ostaje nepromijenjen
	*/

	cout << "KOLEKCIJA OPERATOR ( ): " << endl;
	cout << kolekcijaOperator << endl;


	/* ========== RemoveAt ========== */

	/*
	* metoda RemoveAt treba da ukloni i vrati (kao povratnu vrijednost)
	* clanove kolekcije koji se nalaze na lokaciji definisanoj prvim parametrom.
	*
	* Nakon izvrsenja metode RemoveAt, objekat kolekcijaKopija bi trebao

	* A
	* B
	* D
	* E
	* ....

	* C
	*/

	cout << "KOLEKCIJA REMOVE AT: " << endl;
	cout << kolekcijaRemoveAt << endl;

	cout << "PREOSTALI ELEMENTI: " << endl;
	cout << kolekcijaKopija << endl;


	/* ========== InsertAt ========== */

	/*
	* InsertAt dodaje novi element na lokaciju definisanu prvim parametrom.
	* Metoda vraca novu kolekciju sa umetnutim elementom.
	* U slucaju popunjene kolekcije ili neispravne lokacije potrebno je baciti
	* izuzetak (za potrebe testiranja mozete dodati try catch blok).
	*
	* Nakon izvrsenja, objekat kolekcijaInsertAt bi trebao posjedovati sljedeci sadrzaj:
	* A
	* B
	* X
	* C
	* D
	* E
	* F
	* G
	* H
	* I
	*
	* objekat kolekcijaOriginal ostaje nepromijenjen
	*/
	Kolekcija<char, 10> kolekcijaInsertAt = kolekcijaOriginal.InsertAt(2, 'X');

	try {
		/* InsertAt - neispravna lokacija */
		kolekcijaInsertAt.InsertAt(15, 'Z');
	}
	catch (exception& e) {
		cout << "Exception InsertAt (lokacija): " << e.what() << endl;
	}

	try {
		/* InsertAt - kapacitet popunjen (kolekcijaBrojeva ima 10 elemenata) */
		kolekcijaBrojeva.InsertAt(0, 'Z');
	}
	catch (exception& e) {
		cout << "Exception InsertAt (kapacitet): " << e.what() << endl;
	}

	cout << "KOLEKCIJA INSERT AT: " << endl;
	cout << kolekcijaInsertAt << endl;


	/* ========== RemoveRange ========== */

	/*
	* RemoveRange uklanja broj elemenata definisan drugim parametrom, pocevsi od
