char* GetNizKaraktera(const char* tekst) {
	if (tekst == nullptr)
		return nullptr;
	int vel = strlen(tekst) + 1;
	char* novi = new char[vel];
	strcpy_s(novi, vel, tekst);
	return novi;
}


template<class T, int max>
class Kolekcija { // Dicionary, Rjecnik , Parovi , Kolekcija
public:
	T* _niz;
	int _trenutno;


	Kolekcija() {

		_trenutno = 0;

		_niz = new T[max];

	}


	Kolekcija(const Kolekcija& obj) {

		_trenutno = obj._trenutno;

		_niz = new T[max];

		for (int i = 0; i < _trenutno; i++)
		{
			_niz[i] = obj._niz[i];
		}

	}

	Kolekcija& operator = (const Kolekcija& obj) {
		if (this != &obj) {


			delete[] _niz;


			_trenutno = obj._trenutno;

			_niz = new T[max];

			for (int i = 0; i < _trenutno; i++)
			{
				_niz[i] = obj._niz[i];
			}
		}
		return *this;
	}

	~Kolekcija() {

		delete[] _niz;
		_niz = nullptr;

	}

	friend ostream& operator << (ostream& COUT, const Kolekcija& obj) {


		for (int i = 0; i < obj._trenutno; i++)
		{
			COUT << obj._niz[i] << endl;
		}

		return COUT;
	}



	void AddElement(T el) { // Zivotinja

		// Onemogućiti izlazak niza izvan opsega bacajem izuzetka tipa Exception

		if (max == _trenutno) {

			throw exception("Kapacitet niza je popunjen");

		}

		// Onemogućiti dupliranje elemenata bacajem izuzetka tipa Exception


		for (int i = 0; i < _trenutno; i++)
		{
			// int == int , 66 == 65 , char == char , A == B , Zivotinja == Zivotinja
			if (el == _niz[i]) { // op == izmedju T elemenata --> int == int , char == char , Zivotinja == Zivoitnja

				throw exception("Dupliranje elemenata nije dozvoljeno");

			}
		}


		// Dodaje element tipa T u niz

		_niz[_trenutno] = el;

		_trenutno++;


	}

	//                                     2         5
	Kolekcija operator () (int pocetak, int kraj) {

		Kolekcija kolekcijaOP;  // prazna kolekcija koja u sebi nema nista


		for (int i = pocetak; i <= kraj; i++)
		{
			kolekcijaOP.AddElement(_niz[i]);
		}

		return kolekcijaOP;

	}








	/* metoda RemoveAt treba da ukloni i vrati (kao povratnu vrijednost) clanoverijecnika koji se nalaze na lokaciji definisanoj prvim parametrom.
	* Nakon izvrsenja metode RemoveAt, objekat kolekcijaSlova bi trebao posjedovati sljedeci sadrzaj:
	* A
	* B
	* D
	* E
	* ....
	*
	* a objekat kolekcijaRemoveAt samo element:
	* C
	*/


	Kolekcija RemoveAt(int lokacija) { // alt. Ukloni


		Kolekcija kolekcijaRemoveAt; // prazna kolekcija 

		Kolekcija kolekcijaOriginal; // prazna kolekcija 
		

		for (int i = 0; i < _trenutno; i++)
		{

			if (lokacija == i) {

				kolekcijaRemoveAt.AddElement(_niz[i]);

			}
			else {

				kolekcijaOriginal.AddElement(_niz[i]);


			}


		}


		*this = kolekcijaOriginal;

		return kolekcijaRemoveAt;

	}


	//InsertAt dodaje novi par na lokaciju definisanu prvim parametrom. metoda vraca trenutno stanje kolekcije.
	//u slucaju popunjene kolekcije ili neispravne lokacije potrebno je baciti izuzetak (za potrebe testiranja mozete dodati try catch blok).


	Kolekcija InsertAt(int lokacija, T el) { // alt. DodajNaPoziciju

		if (_trenutno == max || lokacija < 0 || lokacija > _trenutno) {
			throw exception("Lokacija ne postoji");
		}

		Kolekcija kolekcijaInsertAt;

		for (int i = 0; i < _trenutno; i++)
		{
			if (i == lokacija) {
				kolekcijaInsertAt.AddElement(el);
			}
			kolekcijaInsertAt.AddElement(_niz[i]);
		}


		return kolekcijaInsertAt;
	}


	/*UkloniRaspon uklanja broj elemenata definisan drugim parametrom, pocevsi od lokacije
	definisane prvim parametrom (ukljucujuci tu lokaciju). metoda vraca pokazivac na novu kolekciju sa uklonjenim
	elementima, a pozivalac je odgovoran za njenu dealokaciju.*/
	//Kolekcija<char, 10>* kolekcijaRemoveRange = kolekcijaOriginal.RemoveRange(3, 2);

	Kolekcija* RemoveRange(int lokacijaOd, int brojElemenata) {

		if (lokacijaOd < 0 || lokacijaOd + brojElemenata > _trenutno) {
			throw exception("Opseg nije ispravan");
		}

		Kolekcija* kolekcijaRemoveRange = new Kolekcija();

		Kolekcija kolekcijaPreostali;

		for (int i = 0; i < _trenutno; i++)
		{

			//    i >=   3         i < 3 + 2 = 5 ... 3 4  
			if (i >= lokacijaOd && i < lokacijaOd + brojElemenata) {
				kolekcijaRemoveRange->AddElement(_niz[i]);
			}
			else {

				kolekcijaPreostali.AddElement(_niz[i]);

			}
		}

		*this = kolekcijaPreostali;

		return kolekcijaRemoveRange;

	}





};

class Zivotinja {
	char* _naziv;
	int _tezina;
public:

	Zivotinja(const char* naziv = "---", int tezina = 0) {

		_naziv = GetNizKaraktera(naziv);
		_tezina = tezina;

	}

	Zivotinja(const Zivotinja& obj) {

		_naziv = GetNizKaraktera(obj._naziv);
		_tezina = obj._tezina;

	}

	Zivotinja& operator = (const Zivotinja& obj) {
		if (this != &obj) {

			delete[]_naziv;

			_naziv = GetNizKaraktera(obj._naziv);
			_tezina = obj._tezina;

		}
		return *this;
	}

	bool operator == (const Zivotinja& obj) {

		return strcmp(_naziv, obj._naziv) == 0;

	}

	bool operator > (const Zivotinja& obj) {

		return _tezina > obj._tezina;

	}

	~Zivotinja() {

		delete[]_naziv;
		_naziv = nullptr;

	}

	friend ostream& operator << (ostream& COUT, const Zivotinja& obj) {
		COUT << obj._naziv << " [" << obj._tezina << " kg]";
		return COUT;
	}

};
