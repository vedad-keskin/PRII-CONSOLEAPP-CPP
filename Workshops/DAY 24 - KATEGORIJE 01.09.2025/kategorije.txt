//funkcija za generisanje sifre kupca na osnovu imena i prezimena i rednog broja.
//sifra je u formatu INICIJALI:TRENUTNA_GODINA-ID_KUPCA, npr. AB:2025-003.

string GenerisiSifru(const char* imeprezime, int id) {

	string sifra;

	string ip = imeprezime;

	sifra += toupper(imeprezime[0]);
	sifra += toupper(imeprezime[ip.rfind(" ") + 1]);


	sifra += ":";


	time_t utcVrijeme = time(0);

	tm lokalnoVrijeme;

	localtime_s(&lokalnoVrijeme, &utcVrijeme);

	int trenutnaGodina = 1900 + lokalnoVrijeme.tm_year;

	sifra += to_string(trenutnaGodina);

	sifra += "-";


	stringstream ss;

	ss << setfill('0') << setw(3) << id;

	sifra += ss.str();


	return sifra;
}


bool ValidirajSifru(const char* sifra) {

	regex uzorak("[A-Z]{2}[:][0-9]{4}[-][0-9]{3}");

	return regex_match(sifra, uzorak);

}



template<class T1, class T2, int max>
class KolekcijaParova {
	T1* _prvi[max] = { nullptr };
	T2* _drugi[max] = { nullptr };
	int _trenutno = 0;
public:
	KolekcijaParova() = default;

	//KolekcijaParova() : _trenutno(0) {

	//	//_trenutno = 0;
	//	_prvi = nullptr;
	//	_drugi = nullptr;

	//}

	KolekcijaParova(const KolekcijaParova& obj) {

		_trenutno = obj._trenutno;

		for (int i = 0; i < _trenutno; i++)
		{
			_prvi[i] = new T1(*obj._prvi[i]);
			_drugi[i] = new T2(*obj._drugi[i]);
		}
	}

	KolekcijaParova& operator = (const KolekcijaParova& obj) {
		if (this != &obj) {

			for (int i = 0; i < _trenutno; i++) {
				delete _prvi[i];
				delete _drugi[i];
			}

			_trenutno = obj._trenutno;

			for (int i = 0; i < _trenutno; i++)
			{
				_prvi[i] = new T1(*obj._prvi[i]);
				_drugi[i] = new T2(*obj._drugi[i]);
			}
		}
		return *this;
	}


	void Dodaj(T1 el1, T2 el2) {

		if (max == _trenutno) {

			return;

		}

		_prvi[_trenutno] = new T1(el1);
		_drugi[_trenutno] = new T2(el2);

		_trenutno++;

	}

	//DodajNaPoziciju - dodaje par (99, Proizdovi_99) na lokaciju 1 tj.lokaciju definisanu vrijednoscu prvog parametra,
	// a vraca novo stanje kolekcije tj. kolekciju zajedno sa novododatim elementom
	//KolekcijaParova<int, string, 20> prosirenaLista = listaProizvoda.DodajNaPoziciju(1, 99, "Proizvod_99");


	KolekcijaParova DodajNaPoziciju(int lokacija, T1 el1, T2 el2) {

		KolekcijaParova prosirenaLista;

		for (int i = 0; i < _trenutno; i++)
		{

			if (i == lokacija) {

				prosirenaLista.Dodaj(el1, el2);

			}

			prosirenaLista.Dodaj(*_prvi[i], *_drugi[i]);
		}


		return prosirenaLista;
	}
	// UkloniRaspon - od lokacije definisane prvim parametrom uklanja broj elemenata definisanih drugi parametrom
	// (pocevsi od lokacije 2 ukloni 3 elementa), a vraca pokazivac na kolekciju parova s uklonjenim elementima
	//KolekcijaParova<int, string, 20>* uklonjeniProizvodi = prosirenaLista.UkloniRaspon(2, 3);

	KolekcijaParova* UkloniRaspon(int lokacijaOd, int brojElementa) {

		if (lokacijaOd < 0 || lokacijaOd + brojElementa >= _trenutno) {

			throw exception("Neispravan unos");

		}


		KolekcijaParova* uklonjeniProizvodi = new KolekcijaParova();


		KolekcijaParova preostaliProzivodi;

		for (int i = 0; i < _trenutno; i++)
		{

			if (i >= lokacijaOd && i < lokacijaOd + brojElementa) {

				uklonjeniProizvodi->Dodaj(*_prvi[i], *_drugi[i]);

			}
			else {

				preostaliProzivodi.Dodaj(*_prvi[i], *_drugi[i]);

			}

		}

		*this = preostaliProzivodi;

		return uklonjeniProizvodi;
	}

	~KolekcijaParova() {
		for (int i = 0; i < _trenutno; i++) {
			delete _prvi[i]; _prvi[i] = nullptr;
			delete _drugi[i]; _drugi[i] = nullptr;
		}
	}

	int GetTrenutno() const { return _trenutno; }
	T1& GetPrvi(int indeks) const { return *_prvi[indeks]; }
	T2& GetDrugi(int indeks) const { return *_drugi[indeks]; }
	T1& operator[](int indeks) const { return *_prvi[indeks]; }

	int GetTrenutno() { return _trenutno; }
	T1& GetPrvi(int indeks) { return *_prvi[indeks]; }
	T2& GetDrugi(int indeks) { return *_drugi[indeks]; }
	T1& operator[](int indeks) { return *_prvi[indeks]; }

	friend ostream& operator << (ostream& COUT, const KolekcijaParova& obj) {
		for (int i = 0; i < obj._trenutno; i++)
			COUT << obj.GetPrvi(i) << " " << obj.GetDrugi(i) << "\n";
		return COUT;
	}
};

class DatumVrijeme {
	int* _godina, * _mjesec, * _dan, * _sati, * _minute, * _sekunde;
public:

	DatumVrijeme(int dan = 1, int mjesec = 1, int godina = 2000, int sati = 0, int minute = 0, int sekunde = 0) {
		_godina = new int(godina);
		_mjesec = new int(mjesec);
		_dan = new int(dan);
		_sati = new int(sati);
		_minute = new int(minute);
		_sekunde = new int(sekunde);
	}

	DatumVrijeme(const DatumVrijeme& obj) {
		_godina = new int(*obj._godina);
		_mjesec = new int(*obj._mjesec);
		_dan = new int(*obj._dan);
		_sati = new int(*obj._sati);
		_minute = new int(*obj._minute);
		_sekunde = new int(*obj._sekunde);
	}

	DatumVrijeme& operator = (const DatumVrijeme& obj) {
		if (this != &obj) {
			delete _godina; delete _mjesec; delete _dan;
			delete _sati; delete _minute; delete _sekunde;

			_godina = new int(*obj._godina);
			_mjesec = new int(*obj._mjesec);
			_dan = new int(*obj._dan);
			_sati = new int(*obj._sati);
			_minute = new int(*obj._minute);
			_sekunde = new int(*obj._sekunde);
		}
		return *this;
	}

	~DatumVrijeme() {
		delete _godina; delete _mjesec; delete _dan;
		delete _sati; delete _minute; delete _sekunde;

		_godina = nullptr; _mjesec = nullptr; _dan = nullptr;
		_sati = nullptr; _minute = nullptr; _sekunde = nullptr;
	}

	//05.10.2025 09:30:00

	friend ostream& operator << (ostream& COUT, const DatumVrijeme& obj) {
		COUT << setfill('0') << setw(2) << *obj._dan << "." << setw(2) << *obj._mjesec << "." << *obj._godina << " " << setw(2) << *obj._sati << ":" << setw(2) << *obj._minute << ":" << setw(2) << *obj._sekunde;

		return COUT;

	}

	bool operator == (const DatumVrijeme& obj)const {
		return *_godina == *obj._godina && *_mjesec == *obj._mjesec && *_dan == *obj._dan && *_sati == *obj._sati && *_minute == *obj._minute && *_sekunde == *obj._sekunde;

	}
};

class Proizvod {
	char* _naziv;
	Kategorija _kategorija;
	int _cijena;
public:
	Proizvod(const char* naziv = "", Kategorija kategorija = ELEKTRONIKA, int cijena = 0) : _kategorija(kategorija), _cijena(cijena) {

		_naziv = AlocirajTekst(naziv);

	}

	Proizvod(const Proizvod& obj) {

		_naziv = AlocirajTekst(obj._naziv);
		_kategorija = obj._kategorija;
		_cijena = obj._cijena;

	}

	Proizvod& operator = (const Proizvod& obj) {
		if (this != &obj) {
			delete[] _naziv;

			_naziv = AlocirajTekst(obj._naziv);
			_kategorija = obj._kategorija;
			_cijena = obj._cijena;
		}
		return *this;
	}

	bool operator == (const Proizvod& obj) const {
		return strcmp(_naziv, obj._naziv) == 0 && _cijena == obj._cijena && _kategorija == obj._kategorija;
	}

	~Proizvod() {
		delete[] _naziv;
		_naziv = nullptr;
	}

	const char* GetNaziv() const { return _naziv; }
	Kategorija GetKategorija() const { return _kategorija; }
	int GetCijena() const { return _cijena; }

	friend ostream& operator<<(ostream& COUT, const Proizvod& p) {
		COUT << p._naziv << " " << KategorijaNazivi[(int)p._kategorija] << " " << p._cijena;
		return COUT;
	}


};

class Kupac {
	static int _id;
	char* _sifra;
	char* _imePrezime;

	vector<Transakcija*> _transakcije; // Transakcija* ------> Kupovina/Povrat

	// 1. ISPIS -> Info()

	// 2. DODAVANJE -> new Kupovina/Povrat

	// 3. PRISTUP -> dynamic_cast

	// 4. KOPIRANJE -> dynamic_cast

public:
	Kupac(const char* imePrezime) {

		_imePrezime = AlocirajTekst(imePrezime);

		_sifra = AlocirajTekst(GenerisiSifru(imePrezime, _id).c_str());
		_id++;

	}

	Kupac(const Kupac& obj) {

		_sifra = AlocirajTekst(obj._sifra);
		_imePrezime = AlocirajTekst(obj._imePrezime);


		for (int i = 0; i < obj._transakcije.size(); i++)
		{


			Kupovina* castedKupovina = dynamic_cast<Kupovina*>(obj._transakcije[i]); // nullptr

			Povrat* castedPovrat = dynamic_cast<Povrat*>(obj._transakcije[i]);

			if (castedKupovina) {

				_transakcije.push_back(new Kupovina(*castedKupovina));

			}
			else if (castedPovrat) {

				_transakcije.push_back(new Povrat(*castedPovrat));


			}




		}
	}


	Kupac& operator = (const Kupac& obj) {
		if (this != &obj) {

			delete[] _sifra;
			delete[] _imePrezime;

			for (auto* transakcija : _transakcije)
				delete transakcija;

			_transakcije.clear();

			_sifra = AlocirajTekst(obj._sifra);

			_imePrezime = AlocirajTekst(obj._imePrezime);


			for (int i = 0; i < obj._transakcije.size(); i++)
			{


				Kupovina* castedKupovina = dynamic_cast<Kupovina*>(obj._transakcije[i]); // nullptr
				Povrat* castedPovrat = dynamic_cast<Povrat*>(obj._transakcije[i]);

				if (castedKupovina) {

					_transakcije.push_back(new Kupovina(*castedKupovina));
				}
				else if (castedPovrat) {

					_transakcije.push_back(new Povrat(*castedPovrat));


				}




			}
		}
		return *this;
	}


	// DodajTransakciju - oneomguciti dupliranje transakcija sa istim vremenom, kod kupovine onemoguciti
	// dupliranje proizvoda, a povrat omoguciti samo ako je proizvod kupljen.U zavisnosti od rezultata izvrsenja
	// metoda vraca true ili false


	bool DodajTransakciju(Kupovina kupovina) {

		for (int i = 0; i < _transakcije.size(); i++)
		{

			//if (*_transakcije[i] == kupovina) { // TRANSAKCIJE ->  provjeravamo vrijeme

			//	return false;

			//}

			if (_transakcije[i]->GetVrijemeRealizacije() == kupovina.GetVrijemeRealizacije()) { // TRANSAKCIJE ->  provjeravamo vrijeme

				return false;

			}




			// 3. PRISTUP

			// Transakcija* -> Kupovina*

			Kupovina* castedKupovina = dynamic_cast<Kupovina*>(_transakcije[i]);


			if (castedKupovina) {


				for (int j = 0; j < castedKupovina->GetProizvodi().size(); j++) {

					for (int k = 0; k < kupovina.GetProizvodi().size(); k++) {

						if (castedKupovina->GetProizvodi()[j] == kupovina.GetProizvodi()[k]) {

							return false;

						}

					}


				}


			}





		}




		_transakcije.push_back(new Kupovina(kupovina)); // 2. DODAVANJE -> new Kupovina

		return true;

	}


	bool DodajTransakciju(Povrat povrat) {


		for (int i = 0; i < _transakcije.size(); i++)
		{

			//if (_transakcije[i]->GetVrijemeRealizacije() == povrat.GetVrijemeRealizacije()) { // TRANSAKCIJE ->  provjeravamo vrijeme

			//	return false;

			//}



			// Transakcija* -> Kupovina*


			Kupovina* castedKupovina = dynamic_cast<Kupovina*>(_transakcije[i]);


			if (castedKupovina) {


				for (int j = 0; j < castedKupovina->GetProizvodi().size(); j++)
				{

					for (int k = 0; k < povrat.GetProizvodi().size(); k++)
					{

						if (castedKupovina->GetProizvodi()[j] == povrat.GetProizvodi()[k]) {


							_transakcije.push_back(new Povrat(povrat));

							return true;


						}

					}


				}


			}





		}

		return false;


	}


	~Kupac() {

		delete[] _sifra;
		_sifra = nullptr;
		delete[] _imePrezime;
		_imePrezime = nullptr;

		for (auto* transakcija : _transakcije) {
			delete transakcija;
			transakcija = nullptr;
		}

		_transakcije.clear();
	}

	const char* GetSifra() const { return _sifra; }
	const char* GetImePrezime() const { return _imePrezime; }
	vector<Transakcija*>& GetTransakcije() { return _transakcije; }

	friend ostream& operator<<(ostream& COUT, const Kupac& kupac) {
		COUT << crt << kupac._imePrezime << " [" << kupac._sifra << "]\n";

		for (int i = 0; i < kupac._transakcije.size(); i++)
			COUT << " - " << kupac._transakcije[i]->Info() << "\n"; // 1. ISPIS -> Info()

		COUT << crt;
		return COUT;
	}

	bool operator == (const Kupac& obj)const {
		return strcmp(_sifra, obj._sifra) == 0;
	}
};

class Prodavnica {

public:

	Prodavnica(const char* naziv) {
		_naziv = AlocirajTekst(naziv);
	}

	Prodavnica(const Prodavnica& obj) {
		_naziv = AlocirajTekst(obj._naziv);
		_kupci = obj._kupci;
	}

	Prodavnica& operator = (const Prodavnica& obj) {
		if (this != &obj) {
			delete[] _naziv;

			_naziv = AlocirajTekst(obj._naziv);
			_kupci = obj._kupci;

		}
		return *this;
	}

	void DodajKupca(Kupac kupac) {


		for (int i = 0; i < _kupci.size(); i++)
		{
			if (kupac == _kupci[i]) {

				throw exception("Kupac je vec dio odabrane prodavnice");

			}
		}


		_kupci.push_back(kupac);

	}


	bool RegistrujTransakcijuKupcu(const char* sifraKupca, Kupovina kupovina) {


		for (int i = 0; i < _kupci.size(); i++)
		{

			if (strcmp(sifraKupca, _kupci[i].GetSifra()) == 0) {

				return _kupci[i].DodajTransakciju(kupovina); // true i false

			}


		}

		return false;

	}

	////vraca listu svih kupaca iz prodavnice koji su imali najmanje jednu transakciju u proslijedjenoj kategoriji,
	////te koliko su ukupno potrosili na proizvode iz te kategorije (ukupna potrosnja za kategoriju se
	//// izracunava: kupovina - povrat)

	//KolekcijaParova<Kupac, int, 50> potrosnja = tehnika.PotrosnjaPoKategoriji(ELEKTRONIKA);

	//	cout << potrosnja.GetPrvi(i).GetImePrezime() << " [" << potrosnja.GetPrvi(i).GetSifra() << "] " << " potrosio/la " << potrosnja.GetDrugi(i) << " KM na proizvode iz kategorije ELEKTRONIKA" << crt;


	KolekcijaParova<Kupac, int, 50> PotrosnjaPoKategoriji(Kategorija kategorija) {

		KolekcijaParova<Kupac, int, 50> potrosnja;


		// proci kroz sve kupce

		for (int i = 0; i < _kupci.size(); i++)
		{

			int ukupnaPotrosnja = 0;

			bool imaTransakciju = false;

			for (int j = 0; j < _kupci[i].GetTransakcije().size(); j++)
			{

				Kupovina* castedKupovina = dynamic_cast<Kupovina*>(_kupci[i].GetTransakcije()[j]); // nullptr

				Povrat* castedPovrat = dynamic_cast<Povrat*>(_kupci[i].GetTransakcije()[j]);

				if (castedKupovina) {

					for (int k = 0; k < castedKupovina->GetProizvodi().size(); k++)
					{

						if (castedKupovina->GetProizvodi()[k].GetKategorija() == kategorija) {

							ukupnaPotrosnja += castedKupovina->GetProizvodi()[k].GetCijena();

							imaTransakciju = true;

						}


					}

				}
				else if (castedPovrat) {

					for (int k = 0; k < castedPovrat->GetProizvodi().size(); k++)
					{

						if (castedPovrat->GetProizvodi()[k].GetKategorija() == kategorija) {

							ukupnaPotrosnja -= castedPovrat->GetProizvodi()[k].GetCijena();

						}


					}



				}


			}


			if (imaTransakciju) {

				potrosnja.Dodaj(_kupci[i], ukupnaPotrosnja);

			}


		}

		return potrosnja;

	}



	bool operator == (const Prodavnica& obj) {

		return strcmp(_naziv, obj._naziv) == 0;
	}

	~Prodavnica() {
		delete[] _naziv;
		_naziv = nullptr;
	}

	const char* GetNaziv() const { return _naziv; }
	vector<Kupac>& GetKupci() { return _kupci; }

	friend ostream& operator << (ostream& COUT, const Prodavnica& obj) {
		COUT << "Naziv prodavnice: " << obj._naziv << endl;

		for (int i = 0; i < obj._kupci.size(); i++)
		{
			COUT << "\t" << obj._kupci[i] << endl;
		}

		return COUT;
	}

};

bool UcitajPodatke(string nazivFajla, vector<Prodavnica>& prodavnice) {

	ifstream file(nazivFajla);

	if (!file.is_open()) {
		return false;
	}


	string eachRow;

	while (getline(file, eachRow)) {

		int pozicijaCrtice = eachRow.find('|');
		string imePrezime = eachRow.substr(0, pozicijaCrtice);
		string nazivProdavnice = eachRow.substr(pozicijaCrtice + 1);
		Prodavnica novaProdavnica(nazivProdavnice.c_str());
		bool prodavnicaExists = false;

		for (int i = 0; i < prodavnice.size(); i++) {
			if (novaProdavnica == prodavnice[i]) {
				prodavnicaExists = true;
			}
		}
		if (!prodavnicaExists) {
			prodavnice.push_back(novaProdavnica);
		}

		Kupac noviKupac(imePrezime.c_str());
		bool kupacExists = false;
		for (int i = 0; i < prodavnice.size(); i++) {
			if (prodavnice[i] == novaProdavnica) {
				for (int j = 0; j < prodavnice[i].GetKupci().size(); j++) {
					if (prodavnice[i].GetKupci()[j] == noviKupac) {
						kupacExists = true;

					}
				}
				if (!kupacExists) {
					prodavnice[i].DodajKupca(noviKupac);

				}
			}
		}

	}

	return true;



}
